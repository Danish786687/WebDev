function getGrade(score){
    if(score>= 90 && score <=100) return "grade A+";
    if(score>= 80) return "grade A";
    if(score >=70) return "Grade B";
    if(score >= 60) return "Grade c";
    if(score >=33) return "grade d"
    if(score >=0) "fail";
    return "invalid marks entered";
}

//Rock Paper sciccor logic

function rps(user, computer){
    if(user==computer) return "draw";  //same input of computer and user

    if(user=== "rock" && computer==="scissor") return "user";
    if(user==="scissor" && computer === "paper" ) return "user";
    if(user==="paper" && computer==="rock") return "user";

    return "computer"; ///all other cases computer won
}
